-- ============================================================
-- PLATFORM LOGIC
-- ============================================================
local platform
local platformEnabled = false
local rising = false
local riseSpeed = 10
local platformHeight = 5
local platformY = 0
local snapDistance = 30

local function createPlatform()
    if platform then platform:Destroy() end
    platform = Instance.new("Part")
    platform.Name = "FollowPlatform"
    platform.Size = Vector3.new(12, 1, 12)
    platform.Anchored = true
    platform.CanCollide = true
    platform.Material = Enum.Material.SmoothPlastic
    platform.Color = Color3.fromRGB(255, 120, 0)
    local character = player.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    if root then
        platformY = root.Position.Y - platformHeight
        platform.Position = Vector3.new(root.Position.X, platformY, root.Position.Z)
    end
    platform.Parent = workspace
end

local function setPlatformEnabled(state)
    platformEnabled = state
    if platformEnabled then
        createPlatform()
    else
        rising = false
        if platform then platform:Destroy() platform = nil end
    end
end

UserInputService.InputBegan:Connect(function(input, processed)
    if processed then return end
    if input.KeyCode == Enum.KeyCode.R then
        if platformEnabled then rising = not rising end
    end
end)

RunService.Heartbeat:Connect(function(dt)
    if not platformEnabled or not platform or not platform.Parent then return end
    local character = player.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    if not root then return end

    if rising then
        platformY = platformY + riseSpeed * dt
    end

    local targetY = root.Position.Y - platformHeight
    if targetY > platformY then
        platformY = targetY
    end

    if root.Position.Y < platformY - snapDistance then
        if not root:FindFirstChild("OrbitPull") then
            local bv = Instance.new("BodyVelocity")
            bv.Name = "OrbitPull"
            bv.MaxForce = Vector3.new(0, 1e6, 0)
            bv.Velocity = Vector3.new(0, 200, 0)
            bv.Parent = root
            task.delay(0.35, function()
                if bv and bv.Parent then bv:Destroy() end
            end)
        end
    end

    platform.Position = Vector3.new(root.Position.X, platformY, root.Position.Z)
end)

player.CharacterAdded:Connect(function()
    if platformEnabled then task.wait(1) createPlatform() end
end)

local function makeDraggable(frame, handle)
    handle = handle or frame
    local dragging, dragInput, dragStart, startPos
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
        end
    end)
    handle.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        if input ~= dragInput then return end
        local delta = input.Position - dragStart
        frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end
